cmake_minimum_required(VERSION 3.16)
project(arraylib VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
set(CMAKE_BUILD_RPATH "$ORIGIN")
if(MSVC)
    add_compile_options(/W4 /utf-8)
else()
    add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(arraylib SHARED src/arraylib.cpp)
target_compile_definitions(arraylib PRIVATE ARRAYLIB_BUILD)
target_include_directories(arraylib PUBLIC include)
add_library(arraylib_static STATIC src/arraylib.cpp)
target_compile_definitions(arraylib_static PUBLIC ARRAYLIB_STATIC)
target_include_directories(arraylib_static PUBLIC include)
set_target_properties(arraylib_static PROPERTIES OUTPUT_NAME arraylib)
add_executable(app main.cpp)
target_link_libraries(app PRIVATE arraylib)
add_executable(app_static main.cpp)
target_link_libraries(app_static PRIVATE arraylib_static)
include(CTest)
if(BUILD_TESTING)
    foreach(kind unit integration)
        add_executable(test_${kind} tests/test_${kind}.cpp)
        target_link_libraries(test_${kind} PRIVATE arraylib)
        if(MSVC)
            target_compile_options(test_${kind} PRIVATE /UNDEBUG)
        else()
            target_compile_options(test_${kind} PRIVATE -UNDEBUG)
        endif()
        add_test(NAME ${kind} COMMAND test_${kind})
    endforeach()
    add_test(NAME app_smoke COMMAND app)
    set_tests_properties(app_smoke PROPERTIES PASS_REGULAR_EXPRESSION "Median: 1.5")
endif()

add_executable(variant_29 examples/variant_29.cpp)
target_link_libraries(variant_29 PRIVATE arraylib)
if(BUILD_TESTING)
    add_test(NAME variant_29_output
        COMMAND ${CMAKE_COMMAND} -DPROGRAM=$<TARGET_FILE:variant_29>
                -P ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_variant_29.cmake)
endif()
