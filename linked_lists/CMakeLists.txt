cmake_minimum_required(VERSION 3.16)
project(linked_lists LANGUAGES CXX)
add_library(linked_lists lists.cpp)
target_compile_features(linked_lists PUBLIC cxx_std_17)
target_include_directories(linked_lists PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(list_demo main.cpp)
target_link_libraries(list_demo PRIVATE linked_lists)
include(CTest)
if(BUILD_TESTING)
    add_executable(list_tests tests.cpp)
    target_link_libraries(list_tests PRIVATE linked_lists)
    foreach(case singly doubly planets cycle)
        add_test(NAME ${case} COMMAND list_tests ${case})
    endforeach()
endif()
