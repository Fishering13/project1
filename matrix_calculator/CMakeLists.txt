cmake_minimum_required(VERSION 3.20)
project(Matrix_Calculator VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(MSVC)
  add_compile_options(/W4 /utf-8)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
if(ENABLE_SANITIZERS AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(matrix STATIC src/matrix.cpp)
target_include_directories(matrix PUBLIC include)
add_executable(matrix_calculator src/main.cpp)
target_link_libraries(matrix_calculator PRIVATE matrix)
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
  add_test(NAME demo COMMAND matrix_calculator)
  set_tests_properties(demo PROPERTIES PASS_REGULAR_EXPRESSION "Nonzero count: 4")
endif()
find_package(Doxygen QUIET)
if(DOXYGEN_FOUND)
  add_custom_target(documentation COMMAND ${DOXYGEN_EXECUTABLE} Doxyfile
    WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR})
endif()
